cmake_minimum_required(VERSION 3.24)
project(JaiSDL3InstalledConsumer LANGUAGES C CXX)
if(NOT JAI_SDL3_PREFIX OR NOT SDL3_DIR OR NOT JAI_SDL3_PLATFORM)
  message(FATAL_ERROR "An explicit relocated SDL3 SDK prefix/config and target are required")
endif()
find_package(SDL3 3.4.18 EXACT CONFIG REQUIRED COMPONENTS Headers SDL3-static SDL3-shared
             PATHS "${JAI_SDL3_PREFIX}" NO_DEFAULT_PATH)

# Reject any accidental reference to the original staging tree or a system SDK.
function(require_relocated_file path)
  if(NOT EXISTS "${path}")
    message(FATAL_ERROR "Imported SDK path is missing: ${path}")
  endif()
  file(REAL_PATH "${path}" resolved)
  file(REAL_PATH "${JAI_SDL3_PREFIX}" prefix)
  file(RELATIVE_PATH relative "${prefix}" "${resolved}")
  if(IS_ABSOLUTE "${relative}" OR relative MATCHES "^\\.\\.(/|$)")
    message(FATAL_ERROR "Imported path escapes the relocated SDK: ${path}")
  endif()
endfunction()
get_target_property(includes SDL3::Headers INTERFACE_INCLUDE_DIRECTORIES)
foreach(directory IN LISTS includes)
  require_relocated_file("${directory}/SDL3/SDL.h")
endforeach()
if(NOT includes)
  message(FATAL_ERROR "Installed SDL3 header target has no include directory")
endif()
foreach(mode static shared)
  get_target_property(location SDL3::SDL3-${mode} IMPORTED_LOCATION_RELEASE)
  require_relocated_file("${location}")
  get_target_property(kind SDL3::SDL3-${mode} TYPE)
  if(mode STREQUAL "static" AND NOT kind STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "Installed SDL3 static target has the wrong identity")
  elseif(mode STREQUAL "shared" AND NOT kind STREQUAL "SHARED_LIBRARY")
    message(FATAL_ERROR "Installed SDL3 shared target has the wrong identity")
  endif()
  if(WIN32 AND mode STREQUAL "shared")
    get_target_property(implib SDL3::SDL3-shared IMPORTED_IMPLIB_RELEASE)
    require_relocated_file("${implib}")
    get_target_property(archive SDL3::SDL3-static IMPORTED_LOCATION_RELEASE)
    if(implib STREQUAL archive OR NOT location MATCHES "\\.dll$")
      message(FATAL_ERROR "Installed SDL3 import/static/runtime identities were conflated")
    endif()
  endif()
endforeach()

enable_testing()
foreach(language c cpp)
  foreach(mode static shared)
    set(target sdl3-consumer-${language}-${mode})
    add_executable(${target} consumer.${language})
    set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES
                          CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES
                          RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/${target}")
    target_link_libraries(${target} PRIVATE SDL3::SDL3-${mode})
    if(JAI_SDL3_PLATFORM MATCHES "arm64$")
      target_compile_definitions(${target} PRIVATE JAI_CONSUMER_ARM64=1)
    else()
      target_compile_definitions(${target} PRIVATE JAI_CONSUMER_X64=1)
    endif()
    if(WIN32 AND mode STREQUAL "shared")
      add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                         "$<TARGET_FILE:SDL3::SDL3-shared>" "$<TARGET_FILE_DIR:${target}>")
    endif()
    add_test(NAME ${target} COMMAND $<TARGET_FILE:${target}>)
  endforeach()
endforeach()
